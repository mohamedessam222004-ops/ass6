// 0) الدخول للداتابيز
use assignment_6

// 1) Explicit collection with validation rule
db.createCollection("books", {
  validator: {
    $jsonSchema: {
      bsonType: "object",
      required: ["title"],
      properties: {
        title: {
          bsonType: "string",
          minLength: 1,
          description: "title must be a non-empty string"
        }
      }
    }
  }
})

// 2) Implicit collection "authors"
db.authors.insertOne({ name: "Author1", nationality: "British" })

// 3) Capped collection "logs"
db.createCollection("logs", { capped: true, size: 1048576 })

// 4) Index on books.title
db.books.createIndex({ title: 1 })

// 5) Insert one document
db.books.insertOne({
  title: "Book1",
  author: "Ali",
  year: 1937,
  genres: ["Fantasy", "Adventure"]
})

// 6) Insert multiple documents
db.books.insertMany([
  { title: "Future", author: "George Orwell", year: 2020, genres: ["Science Fiction"] },
  { title: "To Kill a Mockingbird", author: "Harper Lee", year: 1960, genres: ["Classic", "Fiction"] },
  { title: "Brave New World", author: "Aldous Huxley", year: 2006, genres: ["Dystopian", "Science Fiction"] }
])

// 7) Insert a new log
db.logs.insertOne({ book_id: "64b5c2d8a123456ef8914", action: "borrowed" })

// 8) Update book "Future" year to 2022
db.books.updateOne({ title: "Future" }, { $set: { year: 2022 } })

// 9) Find book by title
db.books.findOne({ title: "Brave New World" })

// 10) Books between 1990 and 2010
db.books.find({ year: { $gte: 1990, $lte: 2010 } })

// 11) Books that include genre "Science Fiction"
db.books.find({ genres: "Science Fiction" })

// 12) Skip 2, limit 3, sort year desc
db.books.find().sort({ year: -1 }).skip(2).limit(3)

// 13) Year stored as integer
db.books.find({ year: { $type: "int" } })

// 14) Exclude genres Horror & Science Fiction
db.books.find({ genres: { $nin: ["Horror", "Science Fiction"] } })

// 15) Delete books before year 2000
db.books.deleteMany({ year: { $lt: 2000 } })

// 16) Aggregate: books after 2000 sorted desc
db.books.aggregate([
  { $match: { year: { $gt: 2000 } } },
  { $sort: { year: -1 } }
])

// 17) Aggregate: after 2000, only title/author/year
db.books.aggregate([
  { $match: { year: { $gt: 2000 } } },
  { $project: { _id: 0, title: 1, author: 1, year: 1 } }
])

// 18) Aggregate: unwind genres
db.books.aggregate([
  { $unwind: "$genres" },
  { $project: { _id: 0, title: 1, genres: 1 } }
])

// 19) Aggregate: join books with logs
db.logs.aggregate([
  {
    $lookup: {
      from: "books",
      let: { bookIdStr: "$book_id" },
      pipeline: [
        { $match: { $expr: { $eq: [{ $toString: "$_id" }, "$$bookIdStr"] } } },
        { $project: { _id: 0, title: 1, author: 1, year: 1 } }
      ],
      as: "book_details"
    }
  },
  { $project: { _id: 0, action: 1, book_details: 1 } }
])
